------------------------------------------------------------
TABLE: wc_orders (mirror of WooCommerce orders)
------------------------------------------------------------
CREATE TABLE wc_orders (
  id            INTEGER PRIMARY KEY,        -- WC order ID
  number        TEXT NOT NULL,              -- WC order number (display)
  status        TEXT NOT NULL,              -- pending|processing|completed|etc
  customer_email TEXT NOT NULL,
  customer_wc_id INTEGER,                   -- WC customer ID (0 = guest)
  billing_name  TEXT,
  billing_city  TEXT,
  payment_method TEXT,
  total         NUMERIC(10,2) NOT NULL,
  subtotal      NUMERIC(10,2),
  discount      NUMERIC(10,2) DEFAULT 0,
  date_created  TIMESTAMPTZ NOT NULL,
  date_modified TIMESTAMPTZ NOT NULL,
  raw           JSONB,                      -- full WC payload for future fields
  synced_at     TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_wc_orders_email    ON wc_orders(customer_email);
CREATE INDEX idx_wc_orders_status   ON wc_orders(status);
CREATE INDEX idx_wc_orders_date     ON wc_orders(date_created);
CREATE INDEX idx_wc_orders_wc_id    ON wc_orders(customer_wc_id);

------------------------------------------------------------
TABLE: wc_order_items (mirror of WC line items)
------------------------------------------------------------
CREATE TABLE wc_order_items (
  id            SERIAL PRIMARY KEY,
  order_id      INTEGER NOT NULL REFERENCES wc_orders(id) ON DELETE CASCADE,
  product_id    INTEGER NOT NULL,
  product_name  TEXT NOT NULL,
  quantity      INTEGER NOT NULL,
  line_total    NUMERIC(10,2) NOT NULL,
  image_url     TEXT
);

CREATE INDEX idx_wc_order_items_order  ON wc_order_items(order_id);
CREATE INDEX idx_wc_order_items_prod   ON wc_order_items(product_id);

------------------------------------------------------------
TABLE: wc_customers (mirror of WooCommerce customers)
------------------------------------------------------------
CREATE TABLE wc_customers (
  wc_id          INTEGER PRIMARY KEY,
  email          TEXT NOT NULL UNIQUE,
  first_name     TEXT,
  last_name      TEXT,
  total_spent    NUMERIC(10,2) DEFAULT 0,
  orders_count   INTEGER DEFAULT 0,
  date_registered TIMESTAMPTZ,
  synced_at      TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_wc_customers_email ON wc_customers(email);

------------------------------------------------------------
TABLE: rewards (current points balance per customer)
------------------------------------------------------------
CREATE TABLE rewards (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  points      INTEGER NOT NULL DEFAULT 0,
  tier        TEXT NOT NULL DEFAULT 'bronze',  -- bronze|silver|gold|platinum
  updated_at  TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id)
);

------------------------------------------------------------
TABLE: rewards_transactions (points history)
------------------------------------------------------------
CREATE TABLE rewards_transactions (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  points      INTEGER NOT NULL,              -- positive = earn, negative = redeem
  type        TEXT NOT NULL,                 -- 'order_completed'|'prescription'|'signup'|'redemption'
  description TEXT NOT NULL,
  reference   TEXT,                          -- WC order ID or prescription ID
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_rewards_tx_user ON rewards_transactions(user_id);
CREATE INDEX idx_rewards_tx_date ON rewards_transactions(created_at);

------------------------------------------------------------
POINTS RULES
------------------------------------------------------------
Earning:
  - Account signup:            50 pts (once)
  - WC order completed:        1 pt per R1 spent (rounded)
  - Prescription submitted:    25 pts
  - Chronic prescription:      50 pts (isChronic = true)

Tiers (by total lifetime points):
  - Bronze:   0–499 pts
  - Silver:   500–1999 pts
  - Gold:     2000–4999 pts
  - Platinum: 5000+ pts

Redemption (future):
  - 100 pts = R1 discount on next order

Task 1.1 — Webhook receiver
------------------------------------------------------------
File: app/api/webhooks/woocommerce/route.ts

WooCommerce sends POST requests with HMAC-SHA256 signature in header
  X-WC-Webhook-Signature: base64(HMAC(secret, body))

Validate before processing:
  const sig = request.headers.get('x-wc-webhook-signature')
  const body = await request.text()
  const expected = createHmac('sha256', process.env.WC_WEBHOOK_SECRET!)
    .update(body).digest('base64')
  if (sig !== expected) return Response.json({}, { status: 401 })

Parse topic from header X-WC-Webhook-Topic:
  'order.created'   → upsertOrder(payload)
  'order.updated'   → upsertOrder(payload)
  'customer.created'→ upsertCustomer(payload)
  'customer.updated'→ upsertCustomer(payload)

After upsert:
  - If order.status === 'completed': awardOrderPoints(order)
  - If order.status goes to 'completed' from something else: same

Always return 200 immediately — WC retries on non-200.

Task 1.2 — Upsert helpers
------------------------------------------------------------
File: lib/supabase/wc-sync.ts

Uses Supabase service role key (process.env.SUPABASE_SERVICE_ROLE_KEY)
to bypass RLS — webhooks are server-to-server.

upsertOrder(wcOrder: WcWebhookOrder): Promise<void>
  - Map WC fields to wc_orders columns
  - supabase.from('wc_orders').upsert({ id: wcOrder.id, ... }, { onConflict: 'id' })
  - Delete existing wc_order_items for this order_id, re-insert all line_items
  - Upsert into wc_customers if customer_id > 0

upsertCustomer(wcCustomer: WcWebhookCustomer): Promise<void>
  - supabase.from('wc_customers').upsert({ wc_id: wcCustomer.id, ... }, { onConflict: 'wc_id' })

awardOrderPoints(order: WcWebhookOrder): Promise<void>
  - Find Supabase user by email: supabase.auth.admin.listUsers() filtered by email
    OR: SELECT id FROM auth.users WHERE email = order.billing.email
  - If no user found: skip (guest order, no rewards)
  - Points to award: Math.floor(parseFloat(order.total))
  - INSERT into rewards_transactions (type: 'order_completed', reference: order.id.toString())
  - Upsert rewards table (increment points, recalculate tier)

Helper: calculateTier(totalPoints: number): 'bronze'|'silver'|'gold'|'platinum'
  if >= 5000 return 'platinum'
  if >= 2000 return 'gold'
  if >= 500  return 'silver'
  return 'bronze'

Steps:
  - [ ] Define WcWebhookOrder and WcWebhookCustomer TypeScript interfaces
  - [ ] Implement upsertOrder with order_items delete+re-insert
  - [ ] Implement upsertCustomer
  - [ ] Implement awardOrderPoints
  - [ ] Implement calculateTier
  - [ ] Add SUPABASE_SERVICE_ROLE_KEY to .env.local

Task 4.1 — Analytics query functions
------------------------------------------------------------
File: lib/supabase/analytics.ts

All functions use Supabase server client with service role.

getRevenueByPeriod(days: number): Promise<{ date: string, revenue: number }[]>
  SELECT date_trunc('day', date_created) as date, SUM(total) as revenue
  FROM wc_orders
  WHERE status = 'completed'
    AND date_created >= NOW() - INTERVAL '{days} days'
  GROUP BY 1 ORDER BY 1

getOrderStatsByStatus(): Promise<{ status: string, count: number, total: number }[]>
  SELECT status, COUNT(*) as count, SUM(total) as total
  FROM wc_orders GROUP BY status

getTopProducts(limit: number): Promise<{ product_name: string, qty: number, revenue: number }[]>
  SELECT product_name, SUM(quantity) as qty, SUM(line_total) as revenue
  FROM wc_order_items oi
  JOIN wc_orders o ON o.id = oi.order_id
  WHERE o.status = 'completed'
  GROUP BY product_name ORDER BY revenue DESC LIMIT limit

getCustomerGrowth(months: number): Promise<{ month: string, new_customers: number }[]>
  SELECT date_trunc('month', date_registered) as month, COUNT(*) as new_customers
  FROM wc_customers
  WHERE date_registered >= NOW() - INTERVAL '{months} months'
  GROUP BY 1 ORDER BY 1

getPrescriptionMetrics(): Promise<{ status: string, count: number }[]>
  SELECT status, COUNT(*) FROM prescriptions GROUP BY status

getRecentOrders(limit: number): Promise<WcOrderRow[]>
  SELECT * FROM wc_orders ORDER BY date_created DESC LIMIT limit

getRevenueKPIs(): Promise<{ today, this_week, this_month, last_month, growth_pct }>
  Four sub-queries with date filters, compute growth %
