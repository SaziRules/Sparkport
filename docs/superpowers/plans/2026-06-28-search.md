# Search Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add instant product search with a live results dropdown to both desktop and mobile headers, and wire up URL-driven search results on the shop page.

**Architecture:** A Next.js API route delegates queries to WooCommerce's native `search` param; a `useSearch` hook debounces calls with an `AbortController`; a `useSearchKeyboard` hook handles arrow/enter/escape navigation; a `SearchDropdown` renders results inline or absolutely positioned; both headers wire these together; the shop page accepts `?q=` and passes pre-filtered products down to `ShopLayout`.

**Tech Stack:** Next.js 16 App Router, React 19, TypeScript, Tailwind v4, Vitest + @testing-library/react (already installed)

## Global Constraints

- Brand teal: `#009eb9`, dark blue: `#184363`
- Minimum query length before fetching: 2 characters — check `query.trim().length < 2`
- Dropdown shows at most 6 results (`limit=6`)
- Debounce delay: 300ms
- All new components/hooks must have `'use client'` at the top
- Price format: `` `R${p.salePrice.toFixed(2)}` `` — use `product.salePrice` (a number from the `Product` type), NOT `product.price`
- `getProducts()` is exported from `@/lib/wordpress/products` (direct import in the API route)
- `getAllProducts()` is re-exported from `@/lib/wordpress` — add `search?: string` to its params object in `lib/wordpress/products.ts` in Task 7
- Follow existing API route pattern: `NextResponse.json()`, `export const dynamic = 'force-dynamic'`
- No tests for `useSearch` (wraps fetch = integration territory); no RTL component tests — only the API route and keyboard hook get tests
- `SearchResult` type is defined and exported from `lib/hooks/useSearch.ts`; import it from there in `SearchDropdown` and anywhere else it's needed

import { NextResponse } from 'next/server';
import { getProducts } from '@/lib/wordpress/products';

interface SearchResult {
  id: number;
  name: string;
  slug: string;
  price: string;
  image: string;
}

export const dynamic = 'force-dynamic';

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const q = searchParams.get('q') ?? '';
  const limit = Math.min(parseInt(searchParams.get('limit') ?? '6', 10), 6);

  if (q.trim().length < 2) {
    return NextResponse.json({ results: [] }, {
      headers: { 'Cache-Control': 'no-store' },
    });
  }

  try {
    const products = await getProducts({ search: q, per_page: limit });
    const results: SearchResult[] = products.map((p) => ({
      id: p.id,
      name: p.name,
      slug: p.slug,
      price: `R${p.salePrice.toFixed(2)}`,
      image: p.image,
    }));
    return NextResponse.json({ results }, {
      headers: { 'Cache-Control': 'no-store' },
    });
  } catch {
    return NextResponse.json({ results: [] }, {
      headers: { 'Cache-Control': 'no-store' },
    });
  }
}
```

'use client';

import { useState, useCallback } from 'react';

interface KeyboardOptions {
  resultCount: number;
  onSelectHighlighted: () => void;
  onSubmit: () => void;
  onClose: () => void;
}

export function useSearchKeyboard({
  resultCount,
  onSelectHighlighted,
  onSubmit,
  onClose,
}: KeyboardOptions) {
  const [highlightedIndex, setHighlightedIndex] = useState(-1);

  const resetHighlight = useCallback(() => setHighlightedIndex(-1), []);

  function handleKeyDown(e: React.KeyboardEvent) {
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setHighlightedIndex((i) => Math.min(i + 1, resultCount - 1));
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setHighlightedIndex((i) => Math.max(i - 1, -1));
    } else if (e.key === 'Enter') {
      if (highlightedIndex >= 0) {
        onSelectHighlighted();
      } else {
        onSubmit();
      }
    } else if (e.key === 'Escape') {
      onClose();
    }
  }

  return { highlightedIndex, handleKeyDown, resetHighlight };
}
```

```typescript
'use client';

import { useState, useEffect, useRef } from 'react';

export interface SearchResult {
  id: number;
  name: string;
  slug: string;
  price: string;
  image: string;
}

export function useSearch(query: string): {
  results: SearchResult[];
  isLoading: boolean;
  error: string;
} {
  const [results, setResults] = useState<SearchResult[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState('');
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => {
    if (query.trim().length < 2) {
      setResults([]);
      setIsLoading(false);
      setError('');
      return;
    }

    const timer = setTimeout(() => {
      abortRef.current?.abort();
      const controller = new AbortController();
      abortRef.current = controller;

      setIsLoading(true);
      setError('');

      fetch(`/api/products/search?q=${encodeURIComponent(query)}&limit=6`, {
        signal: controller.signal,
      })
        .then((r) => {
          if (!r.ok) throw new Error('fetch failed');
          return r.json() as Promise<{ results: SearchResult[] }>;
        })
        .then((data) => {
          setResults(data.results);
          setIsLoading(false);
        })
        .catch((err: unknown) => {
          if ((err as Error).name === 'AbortError') return;
          setResults([]);
          setError("Couldn't load results. Please try again.");
          setIsLoading(false);
        });
    }, 300);

    return () => {
      clearTimeout(timer);
      abortRef.current?.abort();
    };
  }, [query]);

  return { results, isLoading, error };
}
```

import Image from 'next/image';
import type { SearchResult } from '@/lib/hooks/useSearch';

interface SearchDropdownProps {
  query: string;
  results: SearchResult[];
  isLoading: boolean;
  error: string;
  highlightedIndex: number;
  onResultClick: (slug: string) => void;
  onSeeAll: () => void;

  query,
  results,
  isLoading,
  error,
  highlightedIndex,
  onResultClick,
  onSeeAll,
  className = 'absolute top-full left-0 right-0 z-50 mt-1',

    <div className={`${className} bg-white rounded-2xl border border-neutral-200 shadow-xl overflow-hidden`}>

        <p className="px-4 py-3 text-sm text-neutral-500">
